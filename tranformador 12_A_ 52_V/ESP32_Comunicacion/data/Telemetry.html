From bash

echo "myValue:1234" | nc -u -w0 teleplot.fr 1128

From python

import socket
sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
sock.sendto("myValue:1234".encode(), ("teleplot.fr", 1128))

Check samples

From SerialPort (e.g. arduino)

#include <Arduino.h>
void setup() {
    Serial.begin(115200);
}

float i=0;
void loop() {
    i+=0.1;
    Serial.print(">sin:"); // Important ">" prefix
    Serial.println(sin(i));
    delay(50);
}

> prefix is required on telemetry from Serial ports

From c++

#include "Teleplot.h"
Teleplot teleplot("teleplot.fr", 1128);
int main(int argc, char* argv[]) {
    teleplot.update("myValue", 1234);
}
